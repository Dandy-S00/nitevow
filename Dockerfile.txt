# Build stage
FROM node:22-bookworm-slim AS build

WORKDIR /app

# Enable the pnpm version declared by package.json
RUN corepack enable

COPY package.json pnpm-lock.yaml ./
COPY patches ./patches

RUN pnpm install --frozen-lockfile

COPY . .

RUN pnpm build

# Remove development-only packages after the production bundle is built
RUN pnpm prune --prod

# Runtime stage
FROM node:22-bookworm-slim AS runtime

ENV NODE_ENV=production
ENV PORT=3000

WORKDIR /app

COPY --from=build /app/package.json ./package.json
COPY --from=build /app/node_modules ./node_modules
COPY --from=build /app/dist ./dist
COPY --from=build /app/drizzle ./drizzle
COPY --from=build /app/drizzle.config.ts ./drizzle.config.ts

EXPOSE 3000

# DATABASE_URL must be supplied at runtime.
# Run migrations separately with: pnpm db:push
CMD ["node", "dist/index.js"]
